\newcounter{rev-enumi}

\newcounter{rev-enumiv}

\newcommand\revcounter[1]{%
	\setcounter{#1}{\inteval{\value{#1}-2}}%
	\stepcounter{rev-#1}%
	\gdef\afterrev{%
		\refstepcounter{rev-#1}%
		\setcounter{rev-#1}{0}}}

\AddEnumerateCounter\revcounter\revcounter{} % the 2nd is dummy

\SetEnumitemKey{revRoman}{%
	start = {\inteval{\getrefnumber{enum-\EnumitemId}-1}},
	label = (\Roman*)\revcounter*,
	ref = (\Roman*),
	after = \afterrev
	\label{enum-\EnumitemId}}

\SetEnumitemKey{reversed}{%
	start = {\inteval{\getrefnumber{enum-\EnumitemId}-1}},
	label = \arabic*.\revcounter*,
	ref = \arabic*.,
	after = \afterrev
	\label{enum-\EnumitemId}}
